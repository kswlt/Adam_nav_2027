cmake_minimum_required(VERSION 3.8)
project(rm_nav_sensors)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(observation_batch INTERFACE)
target_include_directories(observation_batch INTERFACE include)
target_link_libraries(observation_batch INTERFACE Eigen3::Eigen)
add_executable(observation_batch_test test/observation_batch_test.cpp)
target_link_libraries(observation_batch_test observation_batch)
enable_testing()
add_test(NAME observation_batch_test COMMAND observation_batch_test)
install(TARGETS observation_batch_test RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
