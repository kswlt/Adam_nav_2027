cmake_minimum_required(VERSION 3.8)
project(rm_nav_sensors)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(rm_nav_interfaces REQUIRED)
add_library(observation_batch INTERFACE)
target_include_directories(observation_batch INTERFACE include)
target_link_libraries(observation_batch INTERFACE Eigen3::Eigen)
add_executable(observation_batch_test test/observation_batch_test.cpp)
target_link_libraries(observation_batch_test observation_batch)
enable_testing()
add_test(NAME observation_batch_test COMMAND observation_batch_test)
target_compile_options(observation_batch_test PRIVATE -UNDEBUG)
add_executable(mid360_quality_test test/mid360_quality_test.cpp)
target_link_libraries(mid360_quality_test observation_batch)
target_compile_features(mid360_quality_test PRIVATE cxx_std_17)
ament_target_dependencies(mid360_quality_test sensor_msgs)
target_compile_options(mid360_quality_test PRIVATE -UNDEBUG)
add_test(NAME mid360_quality_test COMMAND mid360_quality_test)
add_executable(lio_observation_adapter src/lio_observation_adapter_node.cpp)
add_executable(mid360_cloud_guard src/mid360_cloud_guard_node.cpp)
target_compile_features(mid360_cloud_guard PRIVATE cxx_std_17)
target_link_libraries(mid360_cloud_guard observation_batch)
ament_target_dependencies(mid360_cloud_guard rclcpp sensor_msgs std_msgs)
install(TARGETS mid360_cloud_guard RUNTIME DESTINATION lib/${PROJECT_NAME})
target_link_libraries(lio_observation_adapter observation_batch)
ament_target_dependencies(lio_observation_adapter rclcpp sensor_msgs std_msgs tf2_ros rm_nav_interfaces)
install(TARGETS lio_observation_adapter RUNTIME DESTINATION lib/${PROJECT_NAME})
install(TARGETS observation_batch_test RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_export_include_directories(include)
ament_export_dependencies(Eigen3 sensor_msgs)
ament_package()
