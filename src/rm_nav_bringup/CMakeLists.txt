cmake_minimum_required(VERSION 3.8)
project(rm_nav_bringup)
find_package(ament_cmake REQUIRED)
add_library(nav_supervisor INTERFACE)
target_include_directories(nav_supervisor INTERFACE include)
add_executable(nav_supervisor_test test/nav_supervisor_test.cpp)
target_link_libraries(nav_supervisor_test nav_supervisor)
enable_testing()
add_test(NAME nav_supervisor_test COMMAND nav_supervisor_test)
install(TARGETS nav_supervisor_test RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/task_supervisor.py DESTINATION lib/${PROJECT_NAME} RENAME task_supervisor)
install(PROGRAMS scripts/gimbal_tf.py DESTINATION lib/${PROJECT_NAME} RENAME gimbal_tf)
install(PROGRAMS scripts/pcd_map_publisher.py DESTINATION lib/${PROJECT_NAME} RENAME pcd_map_publisher)
install(PROGRAMS scripts/foxglove_trace_publisher.py DESTINATION lib/${PROJECT_NAME} RENAME foxglove_trace_publisher)
install(PROGRAMS scripts/foxglove_graph_publisher.py DESTINATION lib/${PROJECT_NAME} RENAME foxglove_graph_publisher)
install(PROGRAMS scripts/foxglove_map_status.py DESTINATION lib/${PROJECT_NAME} RENAME foxglove_map_status)
ament_package()
