cmake_minimum_required(VERSION 3.8)
project(rm_nav_control)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(rm_nav_planning REQUIRED)
find_package(rm_nav_interfaces REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_library(omni_pid INTERFACE)
target_include_directories(omni_pid INTERFACE include)
add_executable(omni_pid_adapter src/omni_pid_adapter_node.cpp)
target_link_libraries(omni_pid_adapter omni_pid)
ament_target_dependencies(omni_pid_adapter rclcpp geometry_msgs std_msgs nav_msgs)
add_executable(motion_gate src/motion_gate_node.cpp)
target_compile_features(motion_gate PRIVATE cxx_std_17)
ament_target_dependencies(motion_gate rclcpp geometry_msgs std_msgs)
install(TARGETS motion_gate RUNTIME DESTINATION lib/${PROJECT_NAME})
add_library(command_synthesizer INTERFACE)
target_include_directories(command_synthesizer INTERFACE include)
add_executable(command_synthesizer_test test/command_synthesizer_test.cpp)
target_link_libraries(command_synthesizer_test command_synthesizer)
enable_testing()
add_test(NAME command_synthesizer_test COMMAND command_synthesizer_test)
add_executable(trajectory_tracker_test test/trajectory_tracker_test.cpp)
target_include_directories(trajectory_tracker_test PRIVATE include)
target_compile_features(trajectory_tracker_test PRIVATE cxx_std_17)
ament_target_dependencies(trajectory_tracker_test rm_nav_planning)
target_link_libraries(trajectory_tracker_test rm_nav_planning::timed_trajectory)
target_compile_options(trajectory_tracker_test PRIVATE -UNDEBUG)
add_test(NAME trajectory_tracker_test COMMAND trajectory_tracker_test)
add_executable(trajectory_tracker src/trajectory_tracker_node.cpp)
target_include_directories(trajectory_tracker PRIVATE include)
target_compile_features(trajectory_tracker PRIVATE cxx_std_17)
target_link_libraries(trajectory_tracker rm_nav_planning::timed_trajectory)
ament_target_dependencies(trajectory_tracker rclcpp rm_nav_interfaces geometry_msgs nav_msgs std_msgs)
install(TARGETS trajectory_tracker RUNTIME DESTINATION lib/${PROJECT_NAME})
add_test(NAME trajectory_tracker_regression
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/trajectory_tracker_regression.py $<TARGET_FILE:trajectory_tracker>)
set_tests_properties(trajectory_tracker_regression PROPERTIES TIMEOUT 90)
add_executable(omni_pid_test test/omni_pid_test.cpp)
target_link_libraries(omni_pid_test omni_pid)
add_test(NAME omni_pid_test COMMAND omni_pid_test)
target_compile_options(omni_pid_test PRIVATE -UNDEBUG)
add_test(NAME omni_pid_adapter_regression
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/omni_pid_adapter_regression.py $<TARGET_FILE:omni_pid_adapter>)
set_tests_properties(omni_pid_adapter_regression PROPERTIES TIMEOUT 90)
install(TARGETS motion_gate omni_pid_adapter command_synthesizer_test omni_pid_test RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
