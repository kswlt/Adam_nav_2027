cmake_minimum_required(VERSION 3.8)
project(rm_nav_localization)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rm_nav_registration REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(rm_nav_interfaces REQUIRED)
find_package(action_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav_msgs REQUIRED)

add_library(chassis_resolver src/chassis_resolver.cpp)
target_sources(chassis_resolver PRIVATE src/candidate_region.cpp src/local_submap_builder.cpp)
target_compile_features(chassis_resolver PUBLIC cxx_std_17)
target_include_directories(chassis_resolver PUBLIC include ${rm_nav_registration_INCLUDE_DIRS})
target_link_libraries(chassis_resolver Eigen3::Eigen)
add_executable(frozen_map_matcher src/frozen_map_matcher_node.cpp)
target_link_libraries(frozen_map_matcher rm_nav_registration::small_gicp_backend rm_nav_registration::kiss_gicp_backend chassis_resolver)
ament_target_dependencies(frozen_map_matcher rclcpp sensor_msgs geometry_msgs std_msgs rm_nav_interfaces)
add_executable(map_odom_manager src/map_odom_manager_node.cpp)
target_link_libraries(map_odom_manager chassis_resolver)
ament_target_dependencies(map_odom_manager rclcpp sensor_msgs geometry_msgs std_msgs tf2_ros rm_nav_interfaces action_msgs nav2_msgs nav_msgs)

add_executable(chassis_resolver_test test/chassis_resolver_test.cpp)
target_link_libraries(chassis_resolver_test chassis_resolver)
add_executable(candidate_region_test test/candidate_region_test.cpp)
target_link_libraries(candidate_region_test chassis_resolver)
add_executable(map_odom_manager_test test/map_odom_manager_test.cpp)
target_sources(chassis_resolver PRIVATE src/map_odom_manager.cpp)
target_link_libraries(map_odom_manager_test chassis_resolver)
add_executable(local_submap_builder_test test/local_submap_builder_test.cpp)
target_link_libraries(local_submap_builder_test chassis_resolver)
add_executable(relocalization_state_machine_test test/relocalization_state_machine_test.cpp)
target_link_libraries(relocalization_state_machine_test chassis_resolver)
enable_testing()
add_test(NAME chassis_resolver_test COMMAND chassis_resolver_test)
add_test(NAME candidate_region_test COMMAND candidate_region_test)
add_test(NAME map_odom_manager_test COMMAND map_odom_manager_test)
add_test(NAME local_submap_builder_test COMMAND local_submap_builder_test)
add_test(NAME relocalization_state_machine_test COMMAND relocalization_state_machine_test)
add_executable(ros_conversions_test test/ros_conversions_test.cpp)
target_link_libraries(ros_conversions_test chassis_resolver)
ament_target_dependencies(ros_conversions_test sensor_msgs geometry_msgs rm_nav_interfaces)
add_test(NAME ros_conversions_test COMMAND ros_conversions_test)
foreach(test_target chassis_resolver_test candidate_region_test map_odom_manager_test
                    local_submap_builder_test relocalization_state_machine_test)
  target_compile_options(${test_target} PRIVATE -UNDEBUG)
endforeach()

install(TARGETS chassis_resolver chassis_resolver_test candidate_region_test map_odom_manager_test local_submap_builder_test relocalization_state_machine_test
  ARCHIVE DESTINATION lib
  LIBRARY DESTINATION lib
  RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(TARGETS frozen_map_matcher map_odom_manager RUNTIME DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
