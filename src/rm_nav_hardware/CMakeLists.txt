cmake_minimum_required(VERSION 3.8)
project(rm_nav_hardware)

find_package(ament_cmake REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rm_nav_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_library(rm_serial_hardware_interface SHARED src/rm_serial_hardware_interface.cpp)
target_include_directories(rm_serial_hardware_interface PUBLIC include)
ament_target_dependencies(rm_serial_hardware_interface hardware_interface pluginlib rclcpp rm_nav_interfaces geometry_msgs std_msgs)
target_compile_features(rm_serial_hardware_interface PUBLIC cxx_std_17)
enable_testing()
add_executable(hardware_transport_test test/hardware_transport_test.cpp)
target_link_libraries(hardware_transport_test rm_serial_hardware_interface)
ament_target_dependencies(hardware_transport_test hardware_interface pluginlib rclcpp rm_nav_interfaces geometry_msgs std_msgs)
target_compile_options(hardware_transport_test PRIVATE -UNDEBUG)
add_test(NAME hardware_transport_test COMMAND hardware_transport_test)
set_tests_properties(hardware_transport_test PROPERTIES TIMEOUT 45 ENVIRONMENT "ROS_DOMAIN_ID=197;ROS_AUTOMATIC_DISCOVERY_RANGE=LOCALHOST")
pluginlib_export_plugin_description_file(hardware_interface rm_serial_hardware_interface.xml)
install(TARGETS rm_serial_hardware_interface LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES rm_serial_hardware_interface.xml DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
