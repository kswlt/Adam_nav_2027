cmake_minimum_required(VERSION 3.8)
project(rm_nav_planning)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Boost REQUIRED)
add_library(timed_trajectory src/path_preprocessor.cpp src/swept_footprint_validator.cpp src/minimum_jerk.cpp src/polygon_footprint_validator.cpp)
target_include_directories(timed_trajectory PUBLIC include)
target_include_directories(timed_trajectory SYSTEM PRIVATE ${Boost_INCLUDE_DIRS})
target_link_libraries(timed_trajectory PUBLIC Eigen3::Eigen)
add_executable(timed_trajectory_test test/timed_trajectory_test.cpp)
target_link_libraries(timed_trajectory_test timed_trajectory)
add_executable(path_preprocessor_test test/path_preprocessor_test.cpp)
target_link_libraries(path_preprocessor_test timed_trajectory)
add_executable(swept_footprint_validator_test test/swept_footprint_validator_test.cpp)
target_link_libraries(swept_footprint_validator_test timed_trajectory)
enable_testing()
add_test(NAME timed_trajectory_test COMMAND timed_trajectory_test)
add_test(NAME path_preprocessor_test COMMAND path_preprocessor_test)
add_test(NAME swept_footprint_validator_test COMMAND swept_footprint_validator_test)
add_executable(minimum_jerk_test test/minimum_jerk_test.cpp)
target_link_libraries(minimum_jerk_test timed_trajectory)
add_test(NAME minimum_jerk_test COMMAND minimum_jerk_test)
add_executable(polygon_footprint_validator_test test/polygon_footprint_validator_test.cpp)
target_link_libraries(polygon_footprint_validator_test timed_trajectory)
add_test(NAME polygon_footprint_validator_test COMMAND polygon_footprint_validator_test)
foreach(test_target timed_trajectory_test path_preprocessor_test swept_footprint_validator_test minimum_jerk_test polygon_footprint_validator_test)
  if(MSVC)
    target_compile_options(${test_target} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${test_target} PRIVATE -UNDEBUG)
  endif()
endforeach()
install(TARGETS timed_trajectory path_preprocessor_test timed_trajectory_test swept_footprint_validator_test minimum_jerk_test
  ARCHIVE DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
