cmake_minimum_required(VERSION 3.8)
project(rm_nav_planning)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(timed_trajectory INTERFACE)
target_include_directories(timed_trajectory INTERFACE include)
target_link_libraries(timed_trajectory INTERFACE Eigen3::Eigen)
add_executable(timed_trajectory_test test/timed_trajectory_test.cpp)
target_link_libraries(timed_trajectory_test timed_trajectory)
enable_testing()
add_test(NAME timed_trajectory_test COMMAND timed_trajectory_test)
install(TARGETS timed_trajectory_test RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
