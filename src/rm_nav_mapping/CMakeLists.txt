cmake_minimum_required(VERSION 3.8)
project(rm_nav_mapping)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(rm_nav_interfaces REQUIRED)
find_package(rm_nav_sensors REQUIRED)
find_package(rm_nav_registration REQUIRED)
find_package(GTSAM 4.2.0 EXACT CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)

add_library(keyframe_manager src/keyframe_manager.cpp src/loop_candidate_manager.cpp)
target_compile_features(keyframe_manager PUBLIC cxx_std_17)
target_include_directories(keyframe_manager PUBLIC include)
target_link_libraries(keyframe_manager Eigen3::Eigen)
add_executable(keyframe_manager_test test/keyframe_manager_test.cpp)
target_link_libraries(keyframe_manager_test keyframe_manager)
add_executable(loop_candidate_manager_test test/loop_candidate_manager_test.cpp)
target_link_libraries(loop_candidate_manager_test keyframe_manager)
add_executable(map_bundle_test test/map_bundle_test.cpp)
target_link_libraries(map_bundle_test keyframe_manager)
foreach(test_target keyframe_manager_test loop_candidate_manager_test map_bundle_test)
  target_compile_options(${test_target} PRIVATE -UNDEBUG)
endforeach()
add_executable(keyframe_recorder src/keyframe_recorder_node.cpp)
target_link_libraries(keyframe_recorder keyframe_manager)
ament_target_dependencies(keyframe_recorder rclcpp std_msgs tf2_ros rm_nav_interfaces rm_nav_sensors)
install(TARGETS keyframe_recorder RUNTIME DESTINATION lib/${PROJECT_NAME})
add_library(mapping_pose_graph src/pose_graph.cpp)
target_compile_features(mapping_pose_graph PUBLIC cxx_std_17)
target_include_directories(mapping_pose_graph PUBLIC include ${GTSAM_INCLUDE_DIR})
target_link_libraries(mapping_pose_graph gtsam Eigen3::Eigen)
ament_target_dependencies(mapping_pose_graph rm_nav_registration)
add_executable(pose_graph_test test/pose_graph_test.cpp)
target_link_libraries(pose_graph_test mapping_pose_graph)
add_executable(offline_graph_optimizer src/offline_graph_optimizer.cpp)
target_link_libraries(offline_graph_optimizer mapping_pose_graph OpenSSL::Crypto)
ament_target_dependencies(offline_graph_optimizer rclcpp rm_nav_interfaces rm_nav_sensors rm_nav_registration)
set_target_properties(offline_graph_optimizer PROPERTIES INSTALL_RPATH "$<TARGET_FILE_DIR:gtsam>;$<TARGET_FILE_DIR:small_gicp::small_gicp>")
install(TARGETS offline_graph_optimizer RUNTIME DESTINATION lib/${PROJECT_NAME})
enable_testing()
add_test(NAME pose_graph_test COMMAND pose_graph_test)
add_test(NAME keyframe_manager_test COMMAND keyframe_manager_test)
add_test(NAME loop_candidate_manager_test COMMAND loop_candidate_manager_test)
add_test(NAME map_bundle_test COMMAND map_bundle_test)
install(TARGETS keyframe_manager keyframe_manager_test loop_candidate_manager_test map_bundle_test
  ARCHIVE DESTINATION lib
  RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
